
import express from "express";
import { validationProductMultipleImages } from "../middlewares/validation/product/validationProductMiddleware.js";
import {
  createProduct,
  deleteProductById,
  getProductById,
  getProducts,
  updateProductById,
} from "../controllers/productController.js";

import { validationProductUpdate } from "../middlewares/validation/product/validationProductUpload.js";
import { isValideId } from "../middlewares/isValideId.js";
import multerUplaoder from "../middlewares/upload/multerUploader.js";
import uploadToCloudinary from "../middlewares/upload/uploadToCloudinary.js";
import { checkUserIsAuthenticated } from "../middlewares/auth/protect.js";
import { isAdmin } from "../middlewares/auth/isAdmin.js";

const productRoutes = express.Router();

/*
 * Method : POST
 * Desc : add New product
 * Accessbility : Pivate
 * URL : /add-product
 */

productRoutes.post(
  "/add-product",
  checkUserIsAuthenticated,
  isAdmin,
  multerUplaoder.array("imgs", 8),
  uploadToCloudinary,
  validationProductMultipleImages,
  createProduct,
);

/*
 * Method : GET
 * Desc : add New Parfum
 * Accessbility : Public
 * URL : /all-products
 */

productRoutes.get("/all-products", getProducts);

/*
 * Method : GET
 * Desc : get Product By id
 * Accessbility : Public
 * URL : /:id
 */

productRoutes.get("/:id", isValideId, getProductById);

/*
 * Method : PUT
 * Desc : Update ProductByID
 * Accessbility : Private
 * URL : /:id
 */

productRoutes.put(
  "/:id",
  checkUserIsAuthenticated,
  isAdmin,
  isValideId,
  multerUplaoder.array("imgs", 8),
  uploadToCloudinary,
  validationProductUpdate,
  updateProductById,
);

/*
 * Method : POST
 * Desc : Delete product By Id
 * Accessbility : Private
 * URL : /:id
 */

productRoutes.delete(
  "/:id",
  checkUserIsAuthenticated,
  isAdmin,
  isValideId,
  deleteProductById,
);

export default productRoutes;
import express from "express";

import { Login, Register } from "../controllers/userController.js";

import validationRegister from "../middlewares/validation/user/validationRegister.js";
import multerUplaoder from "../middlewares/upload/multerUploader.js";
import uploadToCloudinary from "../middlewares/upload/uploadToCloudinary.js";
import { isValideAccount } from "../middlewares/auth/isValideAccount.js";
import { validationLogin } from "../middlewares/validation/user/validationLogin.js";
import validationImageMemeType from "../middlewares/validateImgMEMEtype.js";
import { verifyOtp } from "../middlewares/validation/user/verifyOtp.js";
import { EmailTransporter, generateOTP } from "../utils/generateOTP.js";
import User from "../models/User.js";
import { config } from "dotenv";
config();

const userAuthRoutes = express.Router();

/*
 * @route : /register
 * @desc : create new account
 * @accessiblity : public
 * @method : POST
 */
userAuthRoutes.post(
  "/register",
  multerUplaoder.single("img"),
  validationImageMemeType,
  validationRegister,
  uploadToCloudinary,
  Register,
);

// verify Account
userAuthRoutes.post("/verify-otp", verifyOtp, function (req, res) {
  return res.status(200).json({ message: "OTP is Correct" });
});

userAuthRoutes.post(
  "/resend-otp",
  async function (req, res, next) {
    let otpCode = generateOTP();
    let user = await User.findById(req.session.userId);
    await EmailTransporter().sendMail({
      from: process.env.NodeMailer_MAIL,
      to: user.email,
      subject: "Verification OTP Number",
      text: `Your OTP code is ${otpCode}`,
    });

    user.otp = otpCode;
    await user.save();

    next();
  },
  function (req, res) {
    return res.status(200).json({ message: "The OTP has been resended" });
  },
);

// resent OTP code

/*
 * @route : /login
 * @desc : login into my account
 * @accessiblity : public
 * @method : POST
 */
userAuthRoutes.post("/login", validationLogin, isValideAccount, Login);

export default userAuthRoutes;
